使用方式
将剧本制作包保存为文本文件，例如 script.md。

运行脚本：node parse.js script.md（或 node parse.js < script.md）。

脚本会输出一个新的 Markdown 文档，其中 JSON 部分已被替换为可读的表格。

代码
javascript
#!/usr/bin/env node

const fs = require('fs');
const path = require('path');

/**
 * 从文本中提取 ```pro2-production-script ... ``` 围栏内的 JSON
 */
function extractJSON(text) {
  const match = text.match(/```pro2-production-script\n([\s\S]*?)\n```/);
  if (!match) return null;
  try {
    return JSON.parse(match[1]);
  } catch (e) {
    console.error('JSON 解析失败:', e.message);
    return null;
  }
}

/**
 * 将 JSON 对象转换为多个 Markdown 表格的字符串
 */
function jsonToMarkdownTables(obj, titlePrefix = '') {
  if (!obj || typeof obj !== 'object') return '';

  const lines = [];
  const addTable = (header, rows) => {
    if (!rows || rows.length === 0) return;
    lines.push(`\n### ${header}`);
    // 生成 Markdown 表格
    const keys = Object.keys(rows[0]);
    lines.push('| ' + keys.join(' | ') + ' |');
    lines.push('| ' + keys.map(() => '---').join(' | ') + ' |');
    rows.forEach(row => {
      const values = keys.map(k => {
        let val = row[k];
        if (val === undefined || val === null) val = '';
        if (typeof val === 'object') val = JSON.stringify(val);
        return String(val).replace(/\|/g, '\\|'); // 转义管道符
      });
      lines.push('| ' + values.join(' | ') + ' |');
    });
  };

  // 处理顶层字段
  for (const [key, value] of Object.entries(obj)) {
    if (key === 'schemaVersion' || key === 'tier' || key === 'step') continue; // 元信息忽略
    if (key === 'patch') {
      // 处理 patch 内的内容
      const patch = value;
      for (const [subKey, subVal] of Object.entries(patch)) {
        if (subKey === 'meta') {
          // meta 对象转键值对
          const rows = Object.entries(subVal).map(([k, v]) => ({ 字段: k, 值: String(v) }));
          addTable('元数据 (meta)', rows);
        } else if (Array.isArray(subVal)) {
          // 数组转表格
          if (subVal.length === 0) continue;
          // 检查元素是否为对象
          if (typeof subVal[0] === 'object' && subVal[0] !== null) {
            addTable(subKey, subVal);
          } else {
            // 简单数组转单列表格
            const rows = subVal.map(item => ({ 值: String(item) }));
            addTable(subKey, rows);
          }
        } else if (typeof subVal === 'object' && subVal !== null) {
          // 对象转键值对
          const rows = Object.entries(subVal).map(([k, v]) => {
            let val = v;
            if (typeof v === 'object') val = JSON.stringify(v);
            return { 字段: k, 值: String(val) };
          });
          addTable(subKey, rows);
        } else {
          // 简单值，直接显示
          const rows = [{ 字段: subKey, 值: String(subVal) }];
          addTable('', rows);
        }
      }
      break; // 只处理 patch，不处理外层
    }
  }

  return lines.join('\n');
}

/**
 * 解析完整文本，返回 { beforeJSON, jsonData, afterJSON }
 * beforeJSON 为 JSON 围栏之前的文本，afterJSON 为之后的文本（一般无）
 */
function parseProductionScript(text) {
  const match = text.match(/([\s\S]*?)```pro2-production-script\n([\s\S]*?)\n```([\s\S]*)/);
  if (!match) {
    // 没有找到 JSON 围栏，返回原始文本
    return { beforeJSON: text, jsonData: null, afterJSON: '' };
  }
  const before = match[1];
  const jsonStr = match[2];
  const after = match[3] || '';
  let jsonData = null;
  try {
    jsonData = JSON.parse(jsonStr);
  } catch (e) {
    console.error('JSON 解析失败:', e.message);
  }
  return { beforeJSON: before, jsonData, afterJSON: after };
}

/**
 * 将解析结果重新构建为完整的 Markdown 文档（JSON 替换为表格）
 */
function rebuildMarkdown(parsed) {
  let output = parsed.beforeJSON;
  if (parsed.jsonData) {
    const tables = jsonToMarkdownTables(parsed.jsonData);
    if (tables) {
      output += '\n\n## JSON 数据结构（表格化）\n';
      output += tables;
    }
  }
  if (parsed.afterJSON) {
    output += '\n' + parsed.afterJSON;
  }
  return output;
}

/**
 * CLI 入口
 */
function main() {
  let inputText = '';
  const filePath = process.argv[2];

  if (filePath) {
    // 从文件读取
    try {
      inputText = fs.readFileSync(filePath, 'utf-8');
    } catch (e) {
      console.error(`无法读取文件: ${e.message}`);
      process.exit(1);
    }
  } else {
    // 从标准输入读取
    inputText = fs.readFileSync(0, 'utf-8');
  }

  const parsed = parseProductionScript(inputText);
  const output = rebuildMarkdown(parsed);

  // 输出到标准输出
  console.log(output);
}

// 如果直接运行本脚本，执行 CLI
if (require.main === module) {
  main();
}

// 导出解析函数供外部使用
module.exports = {
  parseProductionScript,
  jsonToMarkdownTables,
  rebuildMarkdown
};
说明
解析逻辑：使用正则提取 ````pro2-production-script` 围栏内的 JSON，同时保留前后文本。

表格生成：

对于 JSON 中的 patch 部分，自动识别数组（如 shots、scenes、characters 等）生成多行表格。

对象（如 meta、visualStyle）生成键值对表格。

嵌套对象会以 JSON 字符串形式显示在单元格中，避免无限递归。

输出：生成新的 Markdown 文档，JSON 部分以 ## JSON 数据结构（表格化） 标题开头，后续表格按字段分组展示。

保留原表格：原 Markdown 中的视觉风格总纲、场景辞典等表格保持不变，仅替换 JSON 围栏区域。

示例
假设输入文件为 script.md（即上面的完整制作包），运行命令：

bash
node parse.js script.md > parsed_output.md
